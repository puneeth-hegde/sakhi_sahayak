cmake_minimum_required(VERSION 3.10)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# ================================
# Llama.cpp core library
# ================================
add_subdirectory(llama.cpp)

# ================================
# JNI wrapper (unique name)
# DO NOT name this "llama"
# ================================
add_library(llama_jni SHARED
    llama_wrapper.cpp
)

# ================================
# Include headers for wrapper + llama.cpp
# ================================
target_include_directories(llama_jni PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/include
    ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/src
)

# ================================
# Link llama core library + Android system libs
# Core llama library target name provided by llama.cpp is "llama"
# ================================
target_link_libraries(llama_jni
    llama          # core llama target from add_subdirectory
    android
    log
)

# ================================
# Compile flags
# ================================
target_compile_options(llama_jni PRIVATE
    -fexceptions -funwind-tables -fno-omit-frame-pointer
)
